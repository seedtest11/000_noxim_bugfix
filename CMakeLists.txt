cmake_minimum_required(VERSION 3.16)
project(noxim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(SC_NO_WRITE_CHECK)
file(GLOB_RECURSE core_sources CONFIGURE_DEPENDS src/*.cpp)
list(FILTER core_sources EXCLUDE REGEX "(Main|MockPE|test_buffer_manager)\\.cpp$")
add_library(noxim_core STATIC ${core_sources})
target_link_libraries(noxim_core PUBLIC yaml-cpp systemc)
target_include_directories(noxim_core PUBLIC src)
add_executable(noxim src/Main.cpp)
target_link_libraries(noxim PRIVATE noxim_core)
add_executable(receive_capacity_tests tests/receive_capacity_tests.cpp)
target_link_libraries(receive_capacity_tests PRIVATE noxim_core)

add_executable(run_tests src/smartbuffer/test_buffer_manager.cpp)
file(WRITE "${CMAKE_BINARY_DIR}/systemc_test_entry.cpp" "extern \"C\" int sc_main(int, char**) { return 0; }\n")
target_sources(run_tests PRIVATE "${CMAKE_BINARY_DIR}/systemc_test_entry.cpp")
target_link_libraries(run_tests PRIVATE noxim_core)
